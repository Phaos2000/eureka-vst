set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
	set(CMAKE_OSX_DEPLOYMENT_TARGET "10.13" CACHE STRING "" FORCE)
endif()
# Universal binary: Apple Silicon and Intel Macs (and hosts under Rosetta).
if(APPLE AND NOT CMAKE_OSX_ARCHITECTURES)
	set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "" FORCE)
endif()

option(AMBIKA_BUILD_PLUGIN "Build the JUCE plug-in (needs the JUCE submodule)" ON)

get_filename_component(AMBIKA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(AMBIKA_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")

if(NOT EXISTS "${AMBIKA_ROOT}/avrlib/op.h")
	message(FATAL_ERROR "avrlib is missing. Run: git submodule update --init")
endif()

# Regenerate when the upstream files that feed the generated sources change.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
	"${AMBIKA_ROOT}/avrlib/op.h"
	"${AMBIKA_ROOT}/controller/resources.cc"
	"${AMBIKA_ROOT}/controller/part.cc")

# --- avrlib/op.h without the AVR assembly -----------------------------------
# op.h hardcodes "#define USE_OPTIMIZED_OP" (inline AVR asm). Its #else branch
# has portable C++ versions of every routine; this copy selects that branch.
file(READ "${AMBIKA_ROOT}/avrlib/op.h" AMBIKA_OP_H)
string(REPLACE "#define USE_OPTIMIZED_OP"
	"// USE_OPTIMIZED_OP removed for the plug-in build (portable C++ branch)"
	AMBIKA_OP_H "${AMBIKA_OP_H}")
if(AMBIKA_OP_H MATCHES "\n#define USE_OPTIMIZED_OP")
	message(FATAL_ERROR "Could not disable USE_OPTIMIZED_OP in avrlib/op.h")
endif()
# The portable U24Sub (never compiled upstream) names its result "sum"
# instead of "difference".
string(REPLACE
	"uint32_t difference = av - bv;\n  result.integral = sum >> 8;\n  result.fractional = sum & 0xff;"
	"uint32_t difference = av - bv;\n  result.integral = difference >> 8;\n  result.fractional = difference & 0xff;"
	AMBIKA_OP_H "${AMBIKA_OP_H}")
if(AMBIKA_OP_H MATCHES "result.integral = sum >> 8;\n  result.fractional = sum & 0xff;\n  return result;\n}\n\nstatic inline uint24_t U24ShiftRight")
	message(FATAL_ERROR "Could not patch U24Sub in avrlib/op.h")
endif()
file(WRITE "${AMBIKA_GENERATED_DIR}/op.h.tmp" "${AMBIKA_OP_H}")
configure_file("${AMBIKA_GENERATED_DIR}/op.h.tmp"
	"${AMBIKA_GENERATED_DIR}/avrlib/op.h" COPYONLY)

# --- Tables taken from the controller firmware ------------------------------
function(ambika_extract_table out_var file regex description)
	file(READ "${file}" contents)
	string(REGEX MATCH "${regex}" match "${contents}")
	if(NOT match)
		message(FATAL_ERROR "Could not find ${description} in ${file}")
	endif()
	set(${out_var} "${match}" PARENT_SCOPE)
endfunction()

ambika_extract_table(AMBIKA_LFO_WAVEFORMS "${AMBIKA_ROOT}/controller/resources.cc"
	"const prog_uint8_t wav_res_lfo_waveforms\\[\\] PROGMEM = {[^}]*};"
	"wav_res_lfo_waveforms")
ambika_extract_table(AMBIKA_TICKS_PER_STEP "${AMBIKA_ROOT}/controller/part.cc"
	"static const prog_uint8_t midi_clock_tick_per_step\\[15\\] PROGMEM = {[^}]*};"
	"midi_clock_tick_per_step")
ambika_extract_table(AMBIKA_PHASE_PER_TICK "${AMBIKA_ROOT}/controller/part.cc"
	"static const prog_uint16_t lfo_phase_increment_per_clock_tick\\[15\\] PROGMEM = {[^}]*};"
	"lfo_phase_increment_per_clock_tick")
string(REPLACE "static const" "const" AMBIKA_TICKS_PER_STEP "${AMBIKA_TICKS_PER_STEP}")
string(REPLACE "static const" "const" AMBIKA_PHASE_PER_TICK "${AMBIKA_PHASE_PER_TICK}")
# The last entry is 65536, which wraps to 0 in the firmware's 16-bit table.
string(REGEX REPLACE "65536([ \n]*})" "0 /* 65536 wraps to 0 on the AVR */\\1"
	AMBIKA_PHASE_PER_TICK "${AMBIKA_PHASE_PER_TICK}")

file(WRITE "${AMBIKA_GENERATED_DIR}/controller_tables.cc.tmp"
"// Generated by plugin/CMakeLists.txt from controller/resources.cc and
// controller/part.cc. Do not edit.

#include \"controller_tables.h\"

namespace ambika {

// The controller's LFO wavetables (LFOs 1-3). The voicecard resources only
// contain a 2-byte placeholder under the same name, renamed in this build.
// (Declared extern first: a namespace-scope const would otherwise be local.)
extern const prog_uint8_t wav_res_lfo_waveforms[];
${AMBIKA_LFO_WAVEFORMS}

namespace plugin {

${AMBIKA_TICKS_PER_STEP}

${AMBIKA_PHASE_PER_TICK}

}  // namespace plugin
}  // namespace ambika
")
configure_file("${AMBIKA_GENERATED_DIR}/controller_tables.cc.tmp"
	"${AMBIKA_GENERATED_DIR}/controller_tables.cc" COPYONLY)

# Surge XT filters: sst-filters (header only) with sst-basic-blocks and simde
# (SSE intrinsics on Apple Silicon). Pinned to the commits Surge XT uses.
set(AMBIKA_THIRD_PARTY "${CMAKE_CURRENT_SOURCE_DIR}/third_party")
foreach(lib sst-filters sst-basic-blocks simde)
	if(NOT EXISTS "${AMBIKA_THIRD_PARTY}/${lib}/.git")
		message(FATAL_ERROR "${lib} is missing. Run: git submodule update --init")
	endif()
endforeach()
target_include_directories(ambika_engine SYSTEM PRIVATE
	"${AMBIKA_THIRD_PARTY}/sst-filters/include"
	"${AMBIKA_THIRD_PARTY}/sst-basic-blocks/include"
	"${AMBIKA_THIRD_PARTY}/simde")

# Include order matters: the generated op.h and the shims must win over the
# upstream avrlib headers of the same name.
# SYSTEM: the upstream firmware headers are not warning-clean under the
# plug-in's warning flags.
target_include_directories(ambika_engine SYSTEM PUBLIC
	"${AMBIKA_GENERATED_DIR}"
	"${CMAKE_CURRENT_SOURCE_DIR}/engine/shim"
	"${CMAKE_CURRENT_SOURCE_DIR}/engine"
	"${AMBIKA_ROOT}")
set_target_properties(ambika_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)

# The voicecard resources define a 2-byte stand-in for the LFO wavetables;
# rename it so the controller's full table (controller_tables.cc) is used.
set_source_files_properties("${AMBIKA_ROOT}/voicecard/resources.cc" PROPERTIES
	COMPILE_DEFINITIONS "wav_res_lfo_waveforms=wav_res_lfo_waveforms_voicecard_stub")
# Upstream firmware sources are compiled as-is; their warnings are not ours.
# The generated tables store negative literals in uint8_t arrays, which
# avr-gcc wraps silently; allow the same wrapping here.
if(MSVC)
	set_source_files_properties(${AMBIKA_UPSTREAM_SOURCES} PROPERTIES
		COMPILE_OPTIONS "/w")
else()
	set_source_files_properties(${AMBIKA_UPSTREAM_SOURCES} PROPERTIES
		COMPILE_OPTIONS "-w;-Wno-c++11-narrowing")
endif()
# common/patch.h uses PROGMEM without including <avr/pgmspace.h> (on the AVR it
# always arrives through an earlier include). Make it available everywhere.
if(MSVC)
	target_compile_options(ambika_engine PUBLIC
		"/FI${CMAKE_CURRENT_SOURCE_DIR}/engine/shim/avr/pgmspace.h"
		/utf-8 /bigobj /Zc:__cplusplus /Zc:preprocessor)
	target_compile_definitions(ambika_engine PUBLIC NOMINMAX _USE_MATH_DEFINES)
else()
	target_compile_options(ambika_engine PUBLIC
		"SHELL:-include \"${CMAKE_CURRENT_SOURCE_DIR}/engine/shim/avr/pgmspace.h\"")
endif()

	if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/JUCE/CMakeLists.txt")
		message(FATAL_ERROR "JUCE is missing. Run: git submodule update --init")
	endif()
	add_subdirectory(third_party/JUCE EXCLUDE_FROM_ALL)

	juce_add_plugin(AmbikaPlugin
		PRODUCT_NAME "${AMBIKA_PLUGIN_NAME}"
		COMPANY_NAME "Tootable Instruments"
		BUNDLE_ID "com.tootableinstruments.eureka"
		PLUGIN_MANUFACTURER_CODE Akvs
		PLUGIN_CODE Akv1
		FORMATS VST3 AU Standalone
		IS_SYNTH TRUE
		NEEDS_MIDI_INPUT TRUE
		NEEDS_MIDI_OUTPUT FALSE
		IS_MIDI_EFFECT FALSE
		EDITOR_WANTS_KEYBOARD_FOCUS TRUE
		COPY_PLUGIN_AFTER_BUILD FALSE
		VST3_CATEGORIES Instrument Synth
		AU_MAIN_TYPE kAudioUnitType_MusicDevice)

	# Re-sign each bundle as the very last build step. JUCE signs the VST3 and
	# then adds Contents/Resources/moduleinfo.json, which breaks the seal;
	# hosts that check signatures (Ableton: "a sealed resource is missing or
	# invalid") then refuse to load it. Ad-hoc signing ("-") is enough for
	# local use; use a Developer ID for distribution.
	if(APPLE)
		set(AMBIKA_CODESIGN_IDENTITY "-" CACHE STRING "codesign identity for the plug-in bundles")
		foreach(format VST3 AU Standalone)
			if(TARGET AmbikaPlugin_${format})
				add_custom_command(TARGET AmbikaPlugin_${format} POST_BUILD
					COMMAND codesign --force --deep --sign "${AMBIKA_CODESIGN_IDENTITY}"
						"$<TARGET_BUNDLE_DIR:AmbikaPlugin_${format}>"
					COMMAND codesign --verify --deep --strict
						"$<TARGET_BUNDLE_DIR:AmbikaPlugin_${format}>"
					COMMENT "Signing ${format} bundle"
					VERBATIM)
			endif()
		endforeach()
	endif()

	# The Ambika factory programs (controller/data/programs), compiled in.
	file(GLOB AMBIKA_FACTORY_PROGRAMS CONFIGURE_DEPENDS "${AMBIKA_ROOT}/controller/data/programs/*.PRO")
	juce_add_binary_data(EurekaFactory HEADER_NAME EurekaFactory.h NAMESPACE EurekaFactory
		SOURCES ${AMBIKA_FACTORY_PROGRAMS})
	target_link_libraries(AmbikaPlugin PRIVATE EurekaFactory)

	target_sources(AmbikaPlugin PRIVATE
		source/AmbikaImport.cpp
		source/AmbikaImport.h
		source/PixelGfx.cpp
		source/PixelGfx.h
		source/PluginParameters.cpp
		source/PluginParameters.h
		source/PresetManager.cpp
		source/PresetManager.h
		source/PluginEditor.cpp
		source/PluginEditor.h
		source/PluginProcessor.cpp
		source/PluginProcessor.h)

	target_compile_definitions(AmbikaPlugin PUBLIC
		JUCE_WEB_BROWSER=0
		JUCE_USE_CURL=0
		JUCE_VST3_CAN_REPLACE_VST2=0
		JUCE_DISPLAY_SPLASH_SCREEN=0)

	target_link_libraries(AmbikaPlugin
		PRIVATE
			ambika_engine
			juce::juce_audio_utils
		PUBLIC
			juce::juce_recommended_config_flags
			juce::juce_recommended_lto_flags
			juce::juce_recommended_warning_flags)

	# Host-side test: loads the built VST3 bundle like a DAW would.
	if(BUILD_TESTING)
		juce_add_console_app(ambika_plugin_host_test PRODUCT_NAME "ambika_plugin_host_test")
		target_sources(ambika_plugin_host_test PRIVATE tests/plugin_host_test.cpp)
		target_compile_definitions(ambika_plugin_host_test PRIVATE
			JUCE_PLUGINHOST_VST3=1
			JUCE_WEB_BROWSER=0
			JUCE_USE_CURL=0)
		target_link_libraries(ambika_plugin_host_test PRIVATE
			juce::juce_audio_processors

		add_dependencies(ambika_plugin_host_test AmbikaPlugin_VST3)
		get_target_property(AMBIKA_VST3_DIR AmbikaPlugin_VST3 LIBRARY_OUTPUT_DIRECTORY)
		add_test(NAME ambika_plugin_host_test
			COMMAND ambika_plugin_host_test "${AMBIKA_VST3_DIR}/${AMBIKA_PLUGIN_NAME}.vst3")

		# Preset management tests (in-process, against a temporary folder).
		juce_add_console_app(ambika_preset_test PRODUCT_NAME "ambika_preset_test")

		target_compile_definitions(ambika_preset_test PRIVATE
			JucePlugin_Name="${AMBIKA_PLUGIN_NAME}"
			JUCE_WEB_BROWSER=0

		target_link_libraries(ambika_preset_test PRIVATE

		target_link_libraries(ambika_editor_snapshot PRIVATE

			juce::juce_recommended_config_flags)
